cmake_minimum_required(VERSION 3.25)
project(EM2Recomp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(tools/XenonRecomp)
# Use XenonRecomp's ordinary wrappers for host-side function overrides on Windows.
get_target_property(EM2_RECOMP_DEFINITIONS XenonRecomp COMPILE_DEFINITIONS)
list(REMOVE_ITEM EM2_RECOMP_DEFINITIONS XENON_RECOMP_USE_ALIAS)
set_target_properties(XenonRecomp PROPERTIES COMPILE_DEFINITIONS "${EM2_RECOMP_DEFINITIONS}")
add_executable(em2-inspect src/inspect.cpp)
target_link_libraries(em2-inspect PRIVATE XenonUtils fmt::fmt)
add_executable(em2-disasm src/disasm.cpp)
target_link_libraries(em2-disasm PRIVATE XenonUtils fmt::fmt)
enable_testing()
add_executable(em2-analysis-tests src/analysis_tests.cpp)
target_link_libraries(em2-analysis-tests PRIVATE LibXenonAnalyse)
add_test(NAME bounded-function-analysis COMMAND em2-analysis-tests)
add_executable(em2-lowering-fixture src/lowering_fixture.cpp tools/XenonRecomp/XenonRecomp/recompiler.cpp tools/XenonRecomp/XenonRecomp/recompiler_config.cpp)
target_include_directories(em2-lowering-fixture PRIVATE tools/XenonRecomp/XenonRecomp)
target_link_libraries(em2-lowering-fixture PRIVATE LibXenonAnalyse XenonUtils fmt::fmt tomlplusplus::tomlplusplus xxHash::xxhash)
target_compile_definitions(em2-lowering-fixture PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(em2-lowering-fixture PRIVATE -Wno-switch -Wno-null-arithmetic)
target_precompile_headers(em2-lowering-fixture PRIVATE tools/XenonRecomp/XenonRecomp/pch.h)
